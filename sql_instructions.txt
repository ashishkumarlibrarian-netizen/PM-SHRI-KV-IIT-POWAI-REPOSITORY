-- Run this SQL in your Supabase SQL Editor to support the new Thought of the Day features:

-- 1. Create thoughts table if it doesn't exist
CREATE TABLE IF NOT EXISTS public.thoughts (
  id uuid primary key default gen_random_uuid(),
  title text not null,
  thought text not null,
  author text,
  is_active boolean default false,
  icon text,
  bg_color text,
  text_color text,
  border_color text,
  gradient_start text,
  gradient_end text,
  display_order integer default 0,
  created_at timestamptz default now()
);
